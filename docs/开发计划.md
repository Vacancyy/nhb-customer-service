### 1.2 数据库初始化脚本

**文件**: `init-db/01-init-schema.sql`

```sql
-- 启用 pgvector 扩展
CREATE EXTENSION IF NOT EXISTS vector;

-- 知识库表
CREATE TABLE IF NOT EXISTS knowledge_entries (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  topic VARCHAR(100) NOT NULL,
  question TEXT NOT NULL,
  embedding VECTOR(1536),
  structured_data JSONB NOT NULL,
  category VARCHAR(50),
  created_at TIMESTAMP DEFAULT NOW(),
  updated_at TIMESTAMP DEFAULT NOW()
);

CREATE INDEX idx_knowledge_embedding ON knowledge_entries
  USING ivfflat (embedding vector_cosine_ops) WITH (lists = 100);
CREATE INDEX idx_knowledge_data ON knowledge_entries USING GIN (structured_data);
CREATE INDEX idx_knowledge_category ON knowledge_entries (category);

-- 会话记录表 (用于审计，非热数据)
CREATE TABLE IF NOT EXISTS conversation_logs (
  id BIGSERIAL PRIMARY KEY,
  session_id VARCHAR(64) NOT NULL,
  user_id VARCHAR(64),
  role VARCHAR(10) NOT NULL,          -- 'user' | 'assistant'
  content TEXT NOT NULL,
  intent_plugin VARCHAR(50),          -- 命中的插件 ID
  period INT,                         -- 当期期数
  metadata JSONB,
  created_at TIMESTAMP DEFAULT NOW()
);
CREATE INDEX idx_conversation_session ON conversation_logs (session_id);
CREATE INDEX idx_conversation_time ON conversation_logs (created_at);

-- 认证会话表
CREATE TABLE IF NOT EXISTS auth_sessions (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  channel VARCHAR(30) NOT NULL,       -- 'wode_nanjing' | 'wechat' | 'h5'
  user_id VARCHAR(64),
  name VARCHAR(50),
  id_card_hash VARCHAR(64),           -- 身份证号哈希（不存明文）
  auth_method VARCHAR(20) NOT NULL,   -- 'TOKEN' | 'FACE' | 'THREE_FACTOR'
  authed BOOLEAN DEFAULT FALSE,
  created_at TIMESTAMP DEFAULT NOW(),
  expires_at TIMESTAMP NOT NULL,
  authenticated_at TIMESTAMP
);
CREATE INDEX idx_auth_sessions_id_card ON auth_sessions (id_card_hash);
CREATE INDEX idx_auth_sessions_expires ON auth_sessions (expires_at);
```

#### Step 5: 编写向量检索（原始 SQL）

**文件**: `apps/web/src/lib/db.ts`

```typescript
import { PrismaClient } from '@prisma/client';

export const prisma = new PrismaClient();

/**
 * 向量相似度检索 (pgvector)
 * Prisma 不支持 vector 类型，用 $queryRaw 执行原生 SQL
 */
export async function vectorSearch(
  embedding: number[],
  limit: number = 5,
): Promise<any[]> {
  return prisma.$queryRaw`
    SELECT id, topic, question, structured_data, category
    FROM knowledge_entries
    ORDER BY embedding <=> ${embedding}::vector
    LIMIT ${limit}
  `;
}
```

#### Step 6: 知识库入库脚本

**文件**: `packages/knowledge/ingest.ts`

```typescript
import { OpenAI } from 'openai';
import { prisma } from '../../apps/web/src/lib/db';

const openai = new OpenAI({ apiKey: process.env.OPENAI_API_KEY });

/**
 * 将知识条目入库:
 * 1. LLM 提取结构化数据 (JSONB)
 * 2. 生成 embedding
 * 3. 写入 PostgreSQL
 */
export async function ingestKnowledgeEntry(
  question: string,
  answers: Record<string, any>,
  topic: string,
  category: string,
) {
  // 1. 生成 embedding (用 JSONB 全文内容作为输入)
  const content = `${question} ${JSON.stringify(answers)}`;
  const embedding = await openai.embeddings.create({
    model: 'text-embedding-3-small',
    input: content,
  });
  const vector = embedding.data[0].embedding;

  // 2. 写入数据库 (用原始 SQL 插入 vector)
  await prisma.$executeRaw`
    INSERT INTO knowledge_entries (id, topic, question, structured_data, category, embedding)
    VALUES (gen_random_uuid(), ${topic}, ${question},
            ${JSON.stringify({ answers, source: 'manual' })}::jsonb,
            ${category},
            ${vector}::vector)
  `;
}
```

#### Step 7: RAG 检索管道

**文件**: `apps/web/src/lib/ai/smart-search.ts`

```typescript
import { OpenAI } from 'openai';
import { vectorSearch } from '../db';

const openai = new OpenAI({ apiKey: process.env.OPENAI_API_KEY });

export async function smartSearch(
  userQuery: string,
  options?: { currentPeriod?: number; topK?: number },
) {
  const topK = options?.topK ?? 5;

  // 1. 用户问题 → embedding
  const embedding = await openai.embeddings.create({
    model: 'text-embedding-3-small',
    input: userQuery,
  });

  // 2. 向量检索
  const results = await vectorSearch(embedding.data[0].embedding, topK);

  // 3. 如果指定了期数，过滤 JSONB 数据
  if (options?.currentPeriod) {
    return results.filter((r: any) =>
      r.structured_data?.answers?.[String(options.currentPeriod)]
    );
  }

  return results;
}
```

#### Step 8: 聊天 API 入口

**文件**: `apps/web/src/app/api/chat/route.ts`

```typescript
import { NextRequest } from 'next/server';
import { OpenAI } from 'openai';
import { smartSearch } from '@/lib/ai/smart-search';
import { getLangfuse } from '@/lib/trace';

const openai = new OpenAI({ apiKey: process.env.OPENAI_API_KEY });

export async function POST(req: NextRequest) {
  const { message, sessionId } = await req.json();
  const langfuse = getLangfuse();

  const trace = langfuse.trace({
    name: 'ninghuibao-chat',
    sessionId,
  });

  // 1. 检索
  const spanSearch = trace.span({ name: 'knowledge-search' });
  const results = await smartSearch(message, { currentPeriod: 2 });
  spanSearch.end({ output: { resultCount: results.length } });

  // 2. 生成回答
  const generation = trace.generation({
    name: 'answer-generation',
    model: 'gpt-4o-mini',
    input: { query: message, context: results },
  });

  const response = await openai.chat.completions.create({
    model: 'gpt-4o-mini',
    messages: [
      { role: 'system', content: buildSystemPrompt(results) },
      { role: 'user', content: message },
    ],
  });

  const answer = response.choices[0].message.content || '';
  generation.end({ output: answer });

  return Response.json({ reply: answer });
}
```

```typescript
import Redis from 'ioredis';

const redis = new Redis(process.env.REDIS_URL!);

export async function getSessionHistory(sessionId: string, limit: number = 10) {
  const key = `session:${sessionId}:history`;
  const messages = await redis.lrange(key, 0, limit - 1);
  return messages.map(m => JSON.parse(m));
}

export async function saveMessage(sessionId: string, role: string, content: string) {
  const key = `session:${sessionId}:history`;
  await redis.lpush(key, JSON.stringify({ role, content }));
  await redis.ltrim(key, 0, 9);  // 保持 10 条
  await redis.expire(key, 86400); // 24h TTL
}

export async function getSessionState(sessionId: string) {
  const key = `session:${sessionId}:state`;
  return await redis.hgetall(key);
}

export async function updateSessionState(sessionId: string, data: Record<string, string>) {
  const key = `session:${sessionId}:state`;
  await redis.hmset(key, data);
  await redis.expire(key, 86400);
}
```

```typescript
import crypto from 'crypto';

export async function verifyWodenToken(token: string) {
  // RSA 私钥解密 (或 AES 解密)
  const privateKey = process.env.WDN_PRIVATE_KEY!;
  const decrypted = crypto.privateDecrypt(
    { key: privateKey, padding: crypto.constants.RSA_PKCS1_OAEP_PADDING },
    Buffer.from(token, 'base64'),
  );

  const payload = JSON.parse(decrypted.toString());

  if (payload.exp < Date.now() / 1000) {
    throw new Error('Token已过期');
  }

  return {
    userId: payload.userId,
    name: payload.name,
    idCard: maskIdCard(payload.idCard),
    channel: 'wode_nanjing',
    authMethod: 'TOKEN',
  };
}

function maskIdCard(idCard: string): string {
  return idCard.slice(0, 6) + '********' + idCard.slice(-4);
}
```

#### Step 3: AuthContext 中间件

**文件**: `apps/web/src/lib/auth/context.ts`

```typescript
import Redis from 'ioredis';
const redis = new Redis(process.env.REDIS_URL!);

export interface AuthContext {
  authSessionId: string;
  authed: boolean;
  userId?: string;
  name?: string;
  idCard?: string;
  channel: string;
  authMethod: string;
  expiresAt?: Date;
}

export async function getAuthContext(authSessionId: string): Promise<AuthContext> {
  if (!authSessionId) {
    return { authSessionId: '', authed: false, channel: 'unknown' };
  }

  const session = await redis.hgetall(`auth:${authSessionId}`);
  if (!session || Object.keys(session).length === 0) {
    return { authSessionId, authed: false, channel: 'unknown' };
  }

  if (new Date(session.expiresAt) < new Date()) {
    return { authSessionId, authed: false, channel: 'unknown' };
  }

  return {
    authSessionId,
    authed: session.authed === 'true',
    userId: session.userId,
    name: session.name,
    idCard: session.idCard,
    channel: session.channel,
    authMethod: session.authMethod,
    expiresAt: new Date(session.expiresAt),
  };
}

export async function createAuthSession(data: {
  channel: string;
  name: string;
  idCard: string;
  authMethod: string;
  userId?: string;
}): Promise<string> {
  const id = crypto.randomUUID();
  const sessionData = {
    ...data,
    authed: 'true',
    createdAt: new Date().toISOString(),
    expiresAt: new Date(Date.now() + 24 * 60 * 60 * 1000).toISOString(), // 24h
  };
  await redis.hmset(`auth:${id}`, sessionData);
  await redis.expire(`auth:${id}`, 86400);
  return id;
}
```

### 5.2 评估脚本

**文件**: `packages/evaluation/run.ts`

```typescript
import { readFileSync } from 'fs';
import { parse } from 'yaml';
import { smartSearch } from '../../apps/web/src/lib/ai/smart-search';
import { OpenAI } from 'openai';

const goldenDataset = parse(readFileSync('datasets/golden-qa.yaml', 'utf-8'));
const openai = new OpenAI();

interface EvalResult {
  id: string;
  passed: boolean;
  score: number;
  reason: string;
}

export async function runEvaluation(): Promise<EvalResult[]> {
  const results: EvalResult[] = [];

  for (const qa of goldenDataset.questions) {
    // 1. 检索
    const searchResults = await smartSearch(qa.question, {
      currentPeriod: qa.metadata.period,
      topK: 3,
    });

    // 2. LLM 生成回答
    const response = await openai.chat.completions.create({
      model: 'gpt-4o-mini',
      messages: [
        { role: 'system', content: buildSystemPrompt(searchResults) },
        { role: 'user', content: qa.question },
      ],
    });

    const generatedAnswer = response.choices[0].message.content!;

    // 3. LLM-as-Judge 评分
    const judge = await openai.chat.completions.create({
      model: 'gpt-4o-mini',
      messages: [
        { role: 'system', content: `你是一名评估专家。请判断生成的回答是否正确回答了用户问题。
期望答案: ${qa.expected_answer}
生成回答: ${generatedAnswer}
请输出 JSON: {"score": 0-1, "passed": boolean, "reason": "理由"}` },
      ],
      response_format: { type: 'json_object' },
    });

    const evaluation = JSON.parse(judge.choices[0].message.content!);

    results.push({
      id: qa.id,
      passed: evaluation.passed,
      score: evaluation.score,
      reason: evaluation.reason,
    });
  }

  // 打印统计
  const passed = results.filter(r => r.passed).length;
  const avgScore = results.reduce((sum, r) => sum + r.score, 0) / results.length;

  console.log(`\n评估结果: ${passed}/${results.length} 通过, 平均分: ${avgScore.toFixed(2)}`);
  return results;
}
```
